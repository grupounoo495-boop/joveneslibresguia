    function rankedModules(text) {

        return { module, score };
      }).filter(item => item.score > 0).sort((a, b) => b.score - a.score).map(item => item.module);
    }

    function updateProgress() {
      const done = new Set();
      let readings = 0;
      let videos = 0;
      for (const module of modules) {
        const hasRead = read(progressKey(module, "read")) === true;
        const hasWatched = read(progressKey(module, "video")) === true;
        readings += Number(hasRead);
        videos += Number(hasWatched);
        if (hasRead && hasWatched) done.add(module.id);
        const { card, content } = views.get(module.id);
        const status = done.has(module.id) ? "✓ Completado" : hasRead || hasWatched ? "En progreso" : "Ver módulo";
        say($("[data-tile-status]", card), status);
        $("[data-check=read]", content).setAttribute("aria-pressed", String(hasRead));
        $("[data-check=video]", content).setAttribute("aria-pressed", String(hasWatched));
      }
      const total = modules.length;
      const percent = Math.round(100 * (readings + videos) / (2 * total));
      say($("#percent"), `${percent}%`);
      $("#meter").setAttribute("aria-valuenow", percent);
      $("#meter span").style.width = `${percent}%`;
      say($("#progress-count"), `${done.size} de ${total} módulos · ${readings} lecturas · ${videos} videos`);
      certificateUnlocked = done.size === total;
      say($("#encouragement"), certificateUnlocked ? "Terminaste la guía. Tu certificado está listo para preparar."
        : percent > 0 ? "Cada paso cuenta. Podés seguir cuando tengas ganas y tiempo."
        : "Podés empezar por lo que más te sirva hoy.");
      $("#badges").innerHTML = badgeList.map(([name, requirement, earned]) =>
        earned(done) ? `<li class="earned" title="Insignia ganada">${name} ✓</li>`
          : `<li title="${requirement}">${name}<span class="sr">. Pendiente. ${requirement}.</span></li>`).join("");
      $("#certificate-form").hidden = !certificateUnlocked;
      say($("#certificate-status"), certificateUnlocked
        ? "¡Completaste los 9 módulos! Elegí el nombre que querés en tu constancia."
        : `Para desbloquearlo te faltan ${total - readings} de 9 lecturas y ${total - videos} de 9 videos por confirmar.`);
      if (!certificateUnlocked) {
        $("#preview").hidden = true;
        $("#preview-note").hidden = true;
        $("#certificate-actions").hidden = true;
      }
    }

    function countdown(show, finish) {
      let interval = 0;
      let deadline = 0;
      let remaining = 0;

      function stop() {
        clearInterval(interval);
        interval = 0;
      }

      function tick() {
        remaining = Math.max(0, deadline - Date.now());
        show(Math.ceil(remaining / 1000));
        if (remaining === 0) {
          stop();
          finish();
        }
      }

      return {
        reset(seconds) {
          stop();
          remaining = seconds * 1000;
          show(seconds);
        },
        start() {
          if (interval || !remaining) return;
          deadline = Date.now() + remaining;
          interval = setInterval(tick, 120);
          tick();
        },
        pause() {
          if (interval) tick();
          stop();
        }
      };
    }

        const minutes = String(Math.floor(seconds / 60)).padStart(2, "0");
        say($(".timer", root), `${minutes}:${String(seconds % 60).padStart(2, "0")}`);
      }, () => {

      });

      function reset(nextMode = mode) {
        mode = nextMode;
        running = false;
        paused = false;
        clock.reset(60 * (mode === "study" ? Number(select.value) : 5));
        say(start, "Empezar");
        say(answer, "");
        $$("[data-mode]", root).forEach(button => button.setAttribute("aria-pressed", String(button.dataset.mode === mode)));
      }

      select.onchange = () => reset("study");
      $("[data-reset]", root).onclick = () => reset();
      $$("[data-mode]", root).forEach(button => button.onclick = () => reset(button.dataset.mode));
      start.onclick = () => {
        if (running) {
